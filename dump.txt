# dump.ps1 â€” concatenate all source files into one txt

$outFile = "dump.txt"
$root    = (Get-Location).Path

# Folders to skip entirely
$skipDirs = @('.git', 'node_modules', 'dist', 'build')

# File extensions to include (add/remove as needed)
$includeExt = @(
    '.html', '.htm',
    '.css', '.scss',
    '.js', '.mjs', '.cjs', '.ts',
    '.json', '.jsonc',
    '.md', '.txt',
    '.ps1', '.sql',
    '.svg'
)

# Files to skip even if the extension matches
$skipFiles = @($outFile, 'dump.ps1', 'package-lock.json')

# Collect
$files = Get-ChildItem -Path $root -Recurse -File |
    Where-Object {
        $dirParts = $_.FullName.Substring($root.Length).Split([IO.Path]::DirectorySeparatorChar)
        ($dirParts | Where-Object { $skipDirs -contains $_ }).Count -eq 0
    } |
    Where-Object { $includeExt -contains $_.Extension.ToLower() } |
    Where-Object { $skipFiles -notcontains $_.Name } |
    Sort-Object FullName

# Write
Remove-Item $outFile -ErrorAction Ignore
$utf8 = New-Object System.Text.UTF8Encoding $false

$sb = New-Object System.Text.StringBuilder
[void]$sb.AppendLine("===== TREE =====")
foreach ($f in $files) {
    [void]$sb.AppendLine($f.FullName.Substring($root.Length + 1))
}
[void]$sb.AppendLine()

foreach ($f in $files) {
    $rel = $f.FullName.Substring($root.Length + 1)
    [void]$sb.AppendLine("===== $rel =====")
    try {
        [void]$sb.AppendLine((Get-Content -LiteralPath $f.FullName -Raw -ErrorAction Stop))
    } catch {
        [void]$sb.AppendLine("<error reading file: $($_.Exception.Message)>")
    }
    [void]$sb.AppendLine()
}

[IO.File]::WriteAllText((Join-Path $root $outFile), $sb.ToString(), $utf8)

Write-Host "Wrote $($files.Count) files to $outFile"

function escapeHtml(s) {
    return String(s).replace(/[&<>"']/g, c => ({
        '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
    }[c]));
}

async function renderUserArea() {
    const area = document.getElementById('user-area');
    if (!area) return;

    const session = await getSession();
    const user = session ? await getUser() : null;

    if (!user) {
        area.innerHTML = '<button type="button" id="login-btn">Log in</button>';
        document.getElementById('login-btn').addEventListener('click', () => {
            window.location.href = 'pages/log-in/login.html';
        });
        return;
    }

    const profile = await getProfile(user.id);
    const name    = profile?.username || user.email?.split('@')[0] || 'User';

function dialogLabelFor(label) {
    if (!label) return 'points';
    const t = String(label).trim();
    if (!t) return 'points';
    if (t.startsWith('/')) return 'score';
    return t;
}

function openPointsDialog(url, name, label, max, onComplete) {
    pendingPoints = { url, max, onComplete };
    pointsTitle.textContent = name;
    pointsLabel.textContent = dialogLabelFor(label);
    pointsInput.value = '';
    if (max != null) pointsInput.max = String(max);
    else pointsInput.removeAttribute('max');
    pointsDialog.showModal();
    pointsInput.focus();
}

pointsCancel.addEventListener('click', () => {
    pendingPoints = null;
    pointsDialog.close();
});

pointsForm.addEventListener('submit', async (e) => {
    e.preventDefault();
    if (!pendingPoints) return;

    const points = parseInt(pointsInput.value, 10);
    if (!Number.isFinite(points) || points < 0) return;

    const { url, max, onComplete } = pendingPoints;
    if (max != null && points > max) {
        alert(`Max for this game is ${max}.`);
        return;
    }

    try {
        await submitCheckIn(url, points);
        pendingPoints = null;
        pointsDialog.close();
        await onComplete?.();
    } catch (err) {
        console.error('check_in failed:', err);
        alert(err.message || 'Failed to save points');
    }
});

async function openFriendsDialog(gameName, gameUrl, label) {
    if (!friendsDialog) return;

    friendsGameTitle.textContent = gameName;
    friendsList.innerHTML = '<li class="friend-row muted">Loadingâ€¦</li>';
    friendsDialog.showModal();

    try {
        const friends = await fetchFriendsForGame(gameUrl);

        if (!friends.length) {
            friendsList.innerHTML = '<li class="friend-row muted">No friends yet.</li>';
            return;
        }

        friends.sort((a, b) => {
            if (a.played !== b.played) return a.played ? -1 : 1;
            if (a.played && b.played) {
                const diff = (b.points ?? 0) - (a.points ?? 0);
                if (diff !== 0) return diff;
            }
            return a.username.localeCompare(b.username);
        });

        friendsList.innerHTML = friends.map(f => {
            const streakClass = f.played ? 'hot' : 'zero';
            const pointsHtml = f.played
                ? `<span class="friend-points">${f.points}${label}</span>`
                : `<span class="friend-points muted">didn't play today</span>`;
            const avatarSrc = f.avatar_url || SITE_ICONS.placeholderUser;
            const avatarHtml = `<img class="friend-avatar" src="${escapeHtml(avatarSrc)}" alt="" onerror="this.onerror=null;this.src='${SITE_ICONS.placeholderUser}'">`;
            const selfBadge = f.isSelf ? ' <span class="friend-self">(you)</span>' : '';
            return `
                <li class="friend-row${f.isSelf ? ' friend-row-self' : ''}">
                    ${avatarHtml}
                    <span class="friend-name">${escapeHtml(f.username)}${selfBadge}</span>
                    ${pointsHtml}
                    <span class="card-streak ${streakClass}">
                        <img class="streak-icon" src="${SITE_ICONS.streak}" alt="">
                        <span class="streak-count">${f.streak}</span>
                    </span>
                </li>
            `;
        }).join('');
    } catch (err) {
        console.error('[friends] fetch failed:', err);
        friendsList.innerHTML = `<li class="friend-row muted">${escapeHtml(err.message || 'Failed to load friends')}</li>`;
    }
}

let selectionFriendsToken = 0;

async function renderSelectionFriends(gameUrl, gameLabel) {
    const container = $('#selection-friends');
    if (!container) return;

    const token = ++selectionFriendsToken;
    container.hidden = true;

    if (!currentUser) return;

        const friends = await fetchFriendsForGame(gameUrl);
        if (token !== selectionFriendsToken) return;

        friends.sort((a, b) => a.username.localeCompare(b.username, undefined, { sensitivity: 'base' }));

        const lbl = (typeof gameLabel === 'string') ? gameLabel : ' pts';

// ---------- catalog validation ----------

function validateCatalog(data, baseURL) {
    if (!data || Array.isArray(data) || typeof data !== 'object') {
        throw new Error('The game catalog must contain named collections.');
    }
    function checkURL(value, allowRelative = false) {
        if (typeof value !== 'string' || !value.trim()) throw new Error('Missing game URL.');
        const url = allowRelative ? new URL(value, baseURL) : new URL(value);
        if (!['http:', 'https:'].includes(url.protocol)) throw new Error('Unsupported URL protocol.');
    }
    function checkItems(items) {
        if (!Array.isArray(items)) throw new Error('A collection must be an array.');
        for (const item of items) {
            if (!item || typeof item !== 'object' || Array.isArray(item) || Object.keys(item).length !== 1) {
                throw new Error('Each catalog entry must have one name.');
            }
            const [name, value] = Object.entries(item)[0];
            if (!name.trim()) throw new Error('A game or group needs a name.');
            if (Array.isArray(value)) checkItems(value);
            else if (typeof value === 'string') checkURL(value);
            else {
                if (!value || typeof value !== 'object') throw new Error('Invalid game entry.');
                checkURL(value.url);
                if (value.thumb) checkURL(value.thumb, true);
            }
        }
    }
    for (const [name, items] of Object.entries(data)) {
        if (!name.trim()) throw new Error('A collection needs a name.');
        checkItems(items);
    }
    return data;
}

async function getData(file) {
    const url = new URL(file, import.meta.url);
    const response = await fetch(url);
    if (!response.ok) throw new Error(`Failed to fetch ${url}: ${response.status}`);
    return validateCatalog(await response.json(), url);
}

import { getSession, updatePassword } from '../../supabase/integration.js';

const form      = document.getElementById('change-password-form');

const confirmEl = document.getElementById('confirm-password');

async function init() {
    const session = await getSession();

    if (!session) {
        info.textContent = 'You need to be logged in to change your password.';
        info.className = 'error';
        form.hidden = true;
        return;
    }

    info.hidden = true;
    form.hidden = false;
    passEl.focus();
}

form.addEventListener('submit', async (e) => {
    e.preventDefault();
    errorEl.textContent = '';

    const pw = passEl.value;
    const confirm = confirmEl.value;

    if (pw.length < 6) {
        errorEl.textContent = 'Password must be at least 6 characters.';
        return;
    }
    if (pw !== confirm) {
        errorEl.textContent = 'Passwords do not match.';
        return;

    const btn = form.querySelector('button[type="submit"]');
    btn.disabled = true;
    try {
        await updatePassword(pw);
        window.location.href = '../profile/profile.html';
    } catch (err) {
        errorEl.textContent = err.message || 'Failed to update password.';
        btn.disabled = false;
    }
});

import {
    getSession, getUser, getProfile,
    updateProfile, searchUsers,
    getFollowing, fetchFollowedProfiles, follow, unfollow, logout,
} from '../../supabase/integration.js';

let currentUser = null;
let following = new Set();

const avatarPreview  = document.getElementById('avatar-preview');
const avatarInput    = document.getElementById('avatar-url');
const usernameInput  = document.getElementById('username');
const saveProfileBtn = document.getElementById('save-profile');
const profileMsg     = document.getElementById('profile-msg');

const searchInput   = document.getElementById('friend-search');
const searchResults = document.getElementById('search-results');
const friendsList   = document.getElementById('friends-list');

const logoutBtn     = document.getElementById('logout-btn');

// ---------- helpers ----------

function escapeHtml(s) {
    return String(s).replace(/[&<>"']/g, c => ({
        '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
    }[c]));
}

function showMsg(el, text, kind = 'ok') {
    el.textContent = text;
    el.className = `msg ${kind}`;
    setTimeout(() => {
        if (el.textContent === text) { el.textContent = ''; el.className = 'msg'; }
    }, 4000);
}

function updateAvatarPreview(url) {
    avatarPreview.onerror = null;   // reset in case previous src failed
    avatarPreview.src = url || SITE_ICONS.placeholderUser;
    avatarPreview.style.display = '';
}

// Fallback if the URL in the src fails to load
avatarPreview.addEventListener('error', () => {
    avatarPreview.onerror = null;
    if (avatarPreview.src !== SITE_ICONS.placeholderUser) {
        avatarPreview.src = SITE_ICONS.placeholderUser;
    }
});

function rowHtml(user, actionLabel, actionClass) {
    const avatarSrc = user.avatar_url || SITE_ICONS.placeholderUser;
    const avatar = `<img class="result-avatar" src="${escapeHtml(avatarSrc)}" alt="" onerror="this.onerror=null;this.src='${SITE_ICONS.placeholderUser}'">`;
    return `
        <li class="result-row">
            ${avatar}
            <span class="result-name">${escapeHtml(user.username)}</span>
            <button type="button"
                    class="result-btn ${actionClass}"
                    data-id="${user.id}">
                ${actionLabel}
            </button>
        </li>`;
}

// ---------- friends list ----------

async function renderFriendsList() {
    friendsList.innerHTML = '<li class="result-muted">Loadingâ€¦</li>';
    try {
        const friends = await fetchFollowedProfiles(currentUser.id);
        if (!friends.length) {
            friendsList.innerHTML = '<li class="result-muted">No friends yet â€” search above to add some.</li>';
            return;
        }
        friendsList.innerHTML = friends
            .map(f => rowHtml(f, 'Unfollow', 'following'))
            .join('');
    } catch (err) {
        friendsList.innerHTML = `<li class="result-muted">${escapeHtml(err.message || 'Failed to load friends')}</li>`;
    }
}

// ---------- init ----------

async function init() {
    const session = await getSession();
    if (!session) {
        window.location.href = '../log-in/login.html';
        return;
    }
    currentUser = await getUser();

    const profile = await getProfile(currentUser.id);
    if (profile) {
        usernameInput.value = profile.username || '';
        avatarInput.value   = profile.avatar_url || '';
        updateAvatarPreview(profile.avatar_url);
    } else {
        updateAvatarPreview(null);
    }

    following = await getFollowing(currentUser.id);
    await renderFriendsList();
}

// ---------- profile ----------

avatarInput.addEventListener('input', () => updateAvatarPreview(avatarInput.value.trim()));

saveProfileBtn.addEventListener('click', async () => {
    const username   = usernameInput.value.trim();
    const avatar_url = avatarInput.value.trim() || null;

    if (username.length < 3 || username.length > 20) {
        showMsg(profileMsg, 'Username must be 3â€“20 characters.', 'err');
        return;
    }

    saveProfileBtn.disabled = true;
    try {
        await updateProfile(currentUser.id, { username, avatar_url });
        showMsg(profileMsg, 'Saved!', 'ok');
    } catch (err) {
        if (err.code === '23505') showMsg(profileMsg, 'Username already taken.', 'err');
        else showMsg(profileMsg, err.message || 'Failed to save.', 'err');
    } finally {
        saveProfileBtn.disabled = false;
    }
});

// ---------- friend search ----------

let searchTimeout = null;
searchInput.addEventListener('input', () => {
    clearTimeout(searchTimeout);
    const q = searchInput.value.trim();
    if (q.length < 2) { searchResults.innerHTML = ''; return; }
    searchTimeout = setTimeout(() => doSearch(q), 300);
});

async function doSearch(query) {
    searchResults.innerHTML = '<li class="result-muted">Searchingâ€¦</li>';
    try {
        const users = await searchUsers(query);
        const others = users.filter(u => u.id !== currentUser.id);
        if (!others.length) {
            searchResults.innerHTML = '<li class="result-muted">No users found.</li>';
            return;
        }
        searchResults.innerHTML = others.map(u => {
            const isFollowing = following.has(u.id);
            return rowHtml(u, isFollowing ? 'Following' : 'Follow', isFollowing ? 'following' : '');
        }).join('');
    } catch (err) {
        searchResults.innerHTML = `<li class="result-muted">${escapeHtml(err.message || 'Search failed')}</li>`;
    }
}

// ---------- follow / unfollow ----------

async function toggleFollow(btn) {
    const id = btn.dataset.id;
    const isFollowing = following.has(id);
    btn.disabled = true;
    try {
        if (isFollowing) {
            await unfollow(currentUser.id, id);
            following.delete(id);
            btn.textContent = 'Follow';
            btn.classList.remove('following');
        } else {
            await follow(currentUser.id, id);
            following.add(id);
            btn.textContent = 'Following';
            btn.classList.add('following');
        }
        await renderFriendsList();
        const q = searchInput.value.trim();
        if (q.length >= 2) await doSearch(q);
    } catch (err) {
        alert(err.message || 'Failed to update follow');
    } finally {
        btn.disabled = false;
    }
}

searchResults.addEventListener('click', (e) => {
    const btn = e.target.closest('.result-btn');
    if (btn) toggleFollow(btn);
});

friendsList.addEventListener('click', (e) => {
    const btn = e.target.closest('.result-btn');
    if (btn) toggleFollow(btn);
});

// ---------- logout ----------

logoutBtn.addEventListener('click', async () => {
    await logout();
    window.location.href = '../../index.html';
});

// ---------- change password ----------

document.getElementById('change-password-btn').addEventListener('click', () => {
    window.location.href = '../change-password/change-password.html';
});

// ---------- go ----------

init();

import { signUp, createProfile, isUsernameAvailable, logout } from '../../supabase/integration.js';

const form      = document.getElementById('sign-in-form');
const usernameEl = document.getElementById('username');
const emailEl   = document.getElementById('email');
const passwordEl = document.getElementById('password');
const confirmEl = document.getElementById('confirm-password');
const errorEl   = document.getElementById('error');
const submitBtn = form.querySelector('button[type="submit"]');

form.addEventListener('submit', async (e) => {
    e.preventDefault();
    errorEl.textContent = '';
    submitBtn.disabled = true;

    const username = usernameEl.value.trim();
    const email    = emailEl.value.trim();
    const password = passwordEl.value;
    const confirm  = confirmEl.value;

    if (!/^[a-zA-Z0-9_]{3,20}$/.test(username)) {
        errorEl.textContent = 'Username: 3â€“20 characters, letters/numbers/underscore only.';
        submitBtn.disabled = false;
        return;
    }
    if (password.length < 6) {
        errorEl.textContent = 'Password must be at least 6 characters.';
        submitBtn.disabled = false;
        return;
    }
    if (password !== confirm) {
        errorEl.textContent = 'Passwords do not match.';
        submitBtn.disabled = false;
        return;
    }

    try {
        // best-effort pre-check â€” the DB unique constraint is the real guard
        const available = await isUsernameAvailable(username);
        if (!available) {
            errorEl.textContent = 'Username already taken.';
            submitBtn.disabled = false;
            return;
        }

        const { user, session } = await signUp(email, password);

        if (!session) {
            // email confirmation required â€” user can't create their profile yet
            errorEl.textContent = 'Check your email to confirm, then log in.';
            return;
        }

        // create the profile row (user is now authenticated)
        try {
            await createProfile(user.id, username);
        } catch (err) {
            // rare race: someone else took the username between the pre-check and now
            if (err.code === '23505') {
                await logout();
                errorEl.textContent = 'That username just got taken. Try a different one (note: your email is now registered).';
            } else {
                errorEl.textContent = err.message || 'Could not create profile.';
            }
            submitBtn.disabled = false;
            return;
        }

        // success â€” go to profile
        window.location.href = '../profile/profile.html';
    } catch (err) {
        errorEl.textContent = err.message || 'Sign up failed.';
        submitBtn.disabled = false;
    }
});

export async function submitCheckIn(gameUrl, points) {
    const { data: { user } } = await supabase.auth.getUser();
    if (!user) throw new Error('Not logged in');

    const { error } = await supabase.from('check_ins').insert({

        throw error;
    }

// Fetches the user stats for games
export async function fetchUserStats(userId) {
    const [checkInsRes, streaksRes] = await Promise.all([
        supabase.from('check_ins')
            .select('game_url, points, checked_on')
            .eq('user_id', userId),
        supabase.from('current_streaks')
            .select('game_url, current_streak')
            .eq('user_id', userId),
    ]);
    if (checkInsRes.error) throw checkInsRes.error;
    if (streaksRes.error)   throw streaksRes.error;

    const d = new Date();
    const today = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;

    const stats = new Map();

    for (const row of checkInsRes.data ?? []) {
        const e = stats.get(row.game_url) ?? { today_points: null, current_streak: 0, done_today: false };
        if (row.checked_on === today) {
            e.today_points = row.points;
            e.done_today   = true;
        }
        stats.set(row.game_url, e);
    }
    for (const row of streaksRes.data ?? []) {
        const e = stats.get(row.game_url) ?? { today_points: null, current_streak: 0, done_today: false };
        e.current_streak = row.current_streak;
        stats.set(row.game_url, e);
    }
    return stats;
}

// Fetch your friends scores
export async function fetchFriendsForGame(gameUrl) {
    const { data: { user } } = await supabase.auth.getUser();
    if (!user) return [];

    // followers of me â†’ friend ids. No FK join, so no schema-cache dependency.
    const { data: follows, error: fErr } = await supabase
        .from('follows')
        .select('followee_id')
        .eq('follower_id', user.id);
    if (fErr) throw fErr;

    const friendIds = follows?.map(f => f.followee_id) ?? [];
    const allIds = [user.id, ...friendIds];   // self + friends

    const d = new Date();
    const today = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;

    const [profilesRes, todayRes, streakRes] = await Promise.all([
        supabase.from('profiles')
            .select('id, username, avatar_url')
            .in('id', allIds),
        supabase.from('check_ins')
            .select('user_id, points')
            .eq('game_url', gameUrl)
            .eq('checked_on', today)
            .in('user_id', allIds),
        supabase.from('current_streaks')
            .select('user_id, current_streak')
            .eq('game_url', gameUrl)
            .in('user_id', allIds),
    ]);
    if (profilesRes.error) throw profilesRes.error;
    if (todayRes.error)    throw todayRes.error;
    if (streakRes.error)   throw streakRes.error;

    const profileMap = new Map((profilesRes.data ?? []).map(p => [p.id, p]));
    const todayMap   = new Map((todayRes.data   ?? []).map(r => [r.user_id, r.points]));
    const streakMap  = new Map((streakRes.data  ?? []).map(r => [r.user_id, r.current_streak]));

    return allIds
        .map(id => {
            const profile = profileMap.get(id);
            if (!profile) return null;
            const played = todayMap.has(id);
            return {
                user_id:    id,
                username:   profile.username,
                avatar_url: profile.avatar_url,
                played,
                points:     played ? todayMap.get(id) : null,
                streak:     streakMap.get(id) ?? 0,
                isSelf:     id === user.id,
            };
        })
        .filter(Boolean);
}

    if (error && error.code !== '23505') throw error;
}

// Unfollow a person

    const { error } = await supabase
        .from('follows')
        .delete()
        .eq('follower_id', followerId)
        .eq('followee_id', followeeId);
    if (error) throw error;
}

// Fetch other people profiles
export async function fetchFollowedProfiles(userId) {
    const { data: follows, error: fErr } = await supabase
        .from('follows')
        .select('followee_id')
        .eq('follower_id', userId);
    if (fErr) throw fErr;
    if (!follows?.length) return [];

    const ids = follows.map(f => f.followee_id);
    const { data: profiles, error: pErr } = await supabase
        .from('profiles')
        .select('id, username, avatar_url')
        .in('id', ids);
    if (pErr) throw pErr;

    const order = new Map(ids.map((id, i) => [id, i]));
    return (profiles ?? []).sort((a, b) => order.get(a.id) - order.get(b.id));
}

    if (error) throw error;

}

// Create the profile row for a freshly-signed-up user
export async function createProfile(userId, username, avatarUrl = null) {
    const { error } = await supabase
        .from('profiles')
        .insert({ id: userId, username, avatar_url: avatarUrl });
    if (error) throw error;
}

// Check if a username is still free (best-effort, pre-flight)
export async function isUsernameAvailable(username) {
    const { data, error } = await supabase
        .from('profiles')
        .select('id')
        .eq('username', username)
        .maybeSingle();
    if (error) throw error;
    return !data;
}
